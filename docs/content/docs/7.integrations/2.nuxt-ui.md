---
title: Nuxt UI
description: Use NotForm with Nuxt UI inputs and form fields.
---

Nuxt UI provides styled inputs and a `UFormField` wrapper for labels and error text. NotForm owns the form state and validation, and Nuxt UI renders the controls.

::note
Component props and emitted events can change between Nuxt UI releases. The mappings below follow the pattern in [Custom components](/docs/integrations/custom-components). If a control behaves differently in your version, check its documented events and bind them explicitly.
::

## Setup

Install both modules and register them.

```ts [nuxt.config.ts]
export default defineNuxtConfig({
  modules: ['@nuxt/ui', '@notform/nuxt'],
})
```

## Use NotForm's form, not UForm

Wrap your controls in NotForm's `Form`, not Nuxt UI's `UForm`. `UForm` has its own validation and submit pipeline. Using both makes two systems compete over the same state.

`UFormField` is safe to use on its own. It lays out the label, description, and error text.

## Text inputs

Pass the first message of `field.errors` to `UFormField`'s `error` prop.

::code-collapse
```vue
<script setup lang="ts">
import { z } from 'zod'

const form = useForm({
  initialValues: { email: '', name: '' },
  onSubmit: (data) => {
    console.log(data)
  },
  schema: z.object({
    email: z.email('Enter a valid email address.'),
    name: z.string().min(1, 'Enter your name.'),
  }),
})
</script>

<template>
  <Form
    :form="form"
    novalidate
    class="space-y-4"
  >
    <Field
      v-slot="{ field }"
      :field="form.field('name')"
    >
      <UFormField
        label="Name"
        :error="field.errors[0]?.message"
      >
        <UInput
          v-model="field.model"
          v-bind="field.events"
          class="inline-full"
        />
      </UFormField>
    </Field>

    <Field
      v-slot="{ field }"
      :field="form.field('email')"
    >
      <UFormField
        label="Email"
        :error="field.errors[0]?.message"
      >
        <UInput
          v-model="field.model"
          v-bind="field.events"
          type="email"
          class="inline-full"
        />
      </UFormField>
    </Field>

    <UButton
      type="submit"
      :loading="form.isSubmitting"
    >
      Submit
    </UButton>
  </Form>
</template>
```
:: 

Nuxt UI renders its own error text from the `error` prop, so you do not need `FieldError` here.

## Selects, checkboxes, and switches

Controls that commit a value use explicit handlers.

::code-collapse
```vue
<template>
  <Field
    v-slot="{ field }"
    :field="form.field('role')"
  >
    <UFormField
      label="Role"
      :error="field.errors[0]?.message"
    >
      <USelect
        v-model="field.model"
        :items="['admin', 'editor', 'viewer']"
        class="inline-full"
        @update:model-value="field.events.onChange"
        @blur="field.events.onBlur"
      />
    </UFormField>
  </Field>

  <Field
    v-slot="{ field }"
    :field="form.field('subscribe')"
  >
    <UCheckbox
      v-model="field.model"
      label="Send me product updates"
      @update:model-value="field.events.onChange"
    />
  </Field>
</template>
```
::

Use `boolean` for checkboxes and switches, and the item value type for selects.

## Arrays

Combine `ArrayField` with Nuxt UI controls as with native ones. Use `item.id` as the key.

::code-collapse
```vue
<template>
  <ArrayField
    v-slot="{ arrayField }"
    :array-field="form.arrayField('emails')"
  >
    <div
      v-for="item in arrayField.items"
      :key="item.id"
      class="flex gap-2"
    >
      <Field
        v-slot="{ field }"
        :field="item"
      >
        <UFormField
          :error="field.errors[0]?.message"
          class="flex-1"
        >
          <UInput
            v-model="field.model"
            v-bind="field.events"
            type="email"
            class="inline-full"
          />
        </UFormField>
      </Field>

      <UButton
        color="neutral"
        variant="ghost"
        icon="lucide:trash"
        @click="arrayField.remove(item.index)"
      />
    </div>

    <UButton
      variant="subtle"
      @click="arrayField.append('')"
    >
      Add email
    </UButton>
  </ArrayField>
</template>
```
::
